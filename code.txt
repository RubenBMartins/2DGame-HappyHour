package com.happyhour.controller;

import com.happyhour.model.GameConfig;
import com.happyhour.model.drink.Drink;
import com.happyhour.model.drink.Glass;

import com.happyhour.model.game.GameSession;
import com.happyhour.view.audio.SoundManager;
import com.happyhour.view.game.DrinkNames;
import com.happyhour.view.game.GameScreen;

import javafx.scene.Scene;
import javafx.scene.input.KeyCode;
import javafx.scene.input.MouseButton;
import javafx.scene.input.MouseEvent;
import javafx.scene.layout.Pane;
import javafx.scene.paint.Color;
import javafx.scene.shape.Rectangle;

// Contrôleur : transforme les clics et les touches du joueur en actions sur le modèle (GameSession)
public class BarController {
    private GameSession session;
    private GameScreen screen;

    public BarController(GameSession session, GameScreen screen) {
        this.session = session;
        this.screen = screen;
    }

    // Crée toutes les zones cliquables : les 6 places, les postes du bar et les bulles de commande
    public void createZones(Pane pane) {
        for (int seat = 0; seat < GameConfig.SEATS; seat++) {
            Rectangle seatBox = new Rectangle(GameScreen.seatX(seat), 390, 160, 290);
            seatBox.setFill(Color.TRANSPARENT);
            setupSeat(seatBox, seat);
            pane.getChildren().add(seatBox);
        }
        createStations(pane);
        createBubbleZones(pane);
    }

    private void setupSeat(Rectangle box, int seat) {
        box.setOnMouseClicked(event -> {
            if (session.isPoliceRaidActive()) {
                return;
            }
            if (event.getButton() == MouseButton.PRIMARY) {
                session.serve(seat);
                if (session.isPoliceRaidActive()) {
                    SoundManager.play("siren");
                }
            } else if (event.getButton() == MouseButton.SECONDARY) {
                if (session.getCustomer(seat) != null) {
                    session.kick(seat);
                    SoundManager.play("kick");
                    screen.showKick(seat);
                }
            }
        });
    }

    private Rectangle createZone(Pane pane, double x, double y, double width, double height, String name) {
        Rectangle zone = new Rectangle(x, y, width, height);
        zone.setFill(Color.web("0xFFFFFF", 0));
        screen.installHover(zone, name);
        pane.getChildren().add(zone);
        return zone;
    }

    private void playSound(String name) {
        if (!session.isPoliceRaidActive()) {
            SoundManager.play(name);
        }
    }

    private void addGlassZone(Pane pane, double x, double y, double width, double height, Glass glass) {
        Rectangle zone = createZone(pane, x, y, width, height, DrinkNames.glassName(glass));
        zone.setOnMouseClicked(event -> {
            if (event.getButton() == MouseButton.PRIMARY) {
                session.chooseGlass(glass);
                playSound("bottle");
            }
        });
    }

    private void addIngredientZone(Pane pane, double x, double y, double width, double height, Ingredient ingredient) {
        Rectangle zone = createZone(pane, x, y, width, height, DrinkNames.ingredientName(ingredient));
        zone.setOnMouseClicked(event -> {
            if (event.getButton() == MouseButton.PRIMARY) {
                session.addIngredient(ingredient);
                if (session.getPreparation().getGlass() != null) {
                    playSound("water");
                }
            }
        });
    }

    private void addTechniqueZone(Pane pane, double x, double y, double width, double height, Technique technique) {
        Rectangle zone = createZone(pane, x, y, width, height, DrinkNames.techniqueName(technique));
        zone.setOnMouseClicked(event -> {
            if (event.getButton() == MouseButton.PRIMARY) {
                session.useTechnique(technique);
                if (session.getPreparation().getGlass() != null) {
                    playSound("bottle");
                }
            }
        });
    }

    private void createStations(Pane pane) {
        addGlassZone(pane, 73, 945, 82, 120, Glass.PINT);
        addGlassZone(pane, 175, 915, 92, 150, Glass.MARTINI);
        addGlassZone(pane, 278, 915, 62, 150, Glass.HIGHBALL);
        addGlassZone(pane, 350, 915, 72, 150, Glass.WINE_GLASS);

        addIngredientZone(pane, 527, 885, 80, 180, Ingredient.WHITE_WINE);
        addIngredientZone(pane, 642, 885, 80, 180, Ingredient.VODKA);
        addIngredientZone(pane, 761, 885, 80, 180, Ingredient.GIN);
        addIngredientZone(pane, 880, 885, 80, 180, Ingredient.TEQUILA);
        addIngredientZone(pane, 1011, 885, 80, 180, Ingredient.TRIPLE_SEC);
        addIngredientZone(pane, 1101, 885, 80, 180, Ingredient.RASPBERRY_SYRUP);
        addIngredientZone(pane, 1190, 885, 80, 180, Ingredient.TONIC);
        addIngredientZone(pane, 1280, 885, 80, 180, Ingredient.LEMON_JUICE);
        addIngredientZone(pane, 1376, 885, 84, 180, Ingredient.CRANBERRY_JUICE);

        addIngredientZone(pane, 54, 540, 234, 277, Ingredient.BEER);
        addIngredientZone(pane, 1506, 882, 446, 178, Ingredient.ICE);
        addIngredientZone(pane, 696, 686, 92, 74, Ingredient.LIME);
        addIngredientZone(pane, 942, 686, 92, 88, Ingredient.ORANGE_ZEST);
        addIngredientZone(pane, 1195, 678, 86, 96, Ingredient.RASPBERRY);

        addTechniqueZone(pane, 1518, 759, 284, 58, Technique.STIR);
        addTechniqueZone(pane, 1799, 552, 107, 211, Technique.SHAKE);
    }

    private void createBubbleZones(Pane pane) {
        for (int seat = 0; seat < GameConfig.SEATS; seat++) {
            int currentSeat = seat;
            Rectangle zone = new Rectangle(GameScreen.seatX(seat) - 10, 170, 180, 220);
            zone.setFill(Color.web("0xFFFFFF", 0));
            zone.addEventHandler(MouseEvent.MOUSE_MOVED, event -> {
                Drink order = session.getOrder(currentSeat);
                if (order != null) {
                    screen.moveHover(order.getName(), event);
                } else {
                    screen.stopHover();
                }
            });
            zone.addEventHandler(MouseEvent.MOUSE_EXITED, event -> screen.stopHover());
            pane.getChildren().add(zone);
        }
    }

    public void installKeys(Scene scene) {
        scene.setOnKeyPressed(event -> {
            if (event.getCode() == KeyCode.SPACE) {
                if (!session.isPoliceRaidActive()) {
                    screen.pause();
                }
            } else if (event.getCode() == KeyCode.E) {
                session.spawnChild();
            } else if (event.getCode() == KeyCode.N) {
                session.skipPhase();
            } else if (event.getCode() == KeyCode.O) {
                session.spawnCustomer();
            } else if (event.getCode() == KeyCode.A) {
                session.countdownSkip();
            }
        });
    }
}

package com.happyhour.controller;

import com.happyhour.model.game.GameSession;
import com.happyhour.view.game.GameScreen;

import javafx.animation.AnimationTimer;

public class GameLoop extends AnimationTimer {
    private long lastTime = 0;
    private GameSession session;
    private GameScreen curscreen;

    public GameLoop(GameSession session, GameScreen gameScreen) {
        this.session = session;
        this.curscreen = gameScreen;
    }

    @Override
    public void handle(long now) {
        if (lastTime == 0) {
            lastTime = now;
            return;
        }
        double deltaTime = (now - lastTime) / 1000000000.0;
        lastTime = now;

        session.update(deltaTime);
        if (session.isOver()) {
            return;
        }
        curscreen.render();
    }
}

package com.happyhour.model.customer;

import com.happyhour.model.GameConfig;
import com.happyhour.model.customer.state.CustomerState;
import com.happyhour.model.customer.state.DrinkingState;
import com.happyhour.model.customer.state.LeftState;
import com.happyhour.model.customer.state.WaitingState;
import com.happyhour.model.drink.Drink;
import com.happyhour.model.drink.Preparation;

public abstract class Customer {
    private Drink order;
    private double patience;
    private double maxPatience;
    private double patienceSpeed = 1;
    private CustomerState state = new WaitingState();
    private Preparation servedPreparation;

    public Customer(Drink order, double patience) {
        this.order = order;
        this.patience = patience;
        this.maxPatience = patience;
    }

    public void update(double deltaTime) {
        state.update(this, deltaTime);
    }

    public void serve() {
        if (canBeServed()) {
            state = new DrinkingState(getDrinkingDuration());
        }
    }

    public void kick() {
        state = new LeftState(true);
    }

    public boolean canBeServed() {
        return state.canBeServed();
    }

    public boolean hasLeft() {
        return state.hasLeft();
    }

    public boolean isLost() {
        return state.isLost();
    }

    public boolean isServedQuickly() {
        return patience > maxPatience / 2;
    }

    public boolean accepts(int grade) {
        return true;
    }

    public boolean isChild() {
        return false;
    }

    public void reducePatience(double amount) {
        patience = patience - amount * patienceSpeed;
    }

    public void setPatienceSpeed(double patienceSpeed) {
        this.patienceSpeed = patienceSpeed;
    }

    public void setState(CustomerState state) {
        this.state = state;
    }

    public double getDrinkingDuration() {
        return GameConfig.DRINKING_TIME;
    }

    public Drink getOrder() {
        return order;
    }

    public void setOrder(Drink order) {
        this.order = order;
    }

    public Preparation getServedPreparation() {
        return servedPreparation;
    }

    public void setServedPreparation(Preparation servedPreparation) {
        this.servedPreparation = servedPreparation;
    }

    public double getPatience() {
        return patience;
    }

    public double getMaxPatience() {
        return maxPatience;
    }

    public abstract double computeTip(double baseTip, double speedBonus);
}

package com.happyhour.model.customer;

import java.util.Random;

import com.happyhour.model.GameConfig;

public class CustomerFactory {
    private Random random = new Random();
    private boolean pickyArrived = false;
    private boolean rushedArrived = false;

    public Customer createCustomer(double elapsed, boolean childrenAllowed) {
        if (elapsed >= GameConfig.FORCE_TIME && !pickyArrived) {
            pickyArrived = true;
            return new PickyCustomer(null);
        }
        if (elapsed >= GameConfig.FORCE_TIME && !rushedArrived) {
            rushedArrived = true;
            return new RushedCustomer(null);
        }

        int roll = random.nextInt(100);
        int limit = GameConfig.NORMAL_RATE;
        if (roll < limit) {
            return new NormalCustomer(null);
        }
        limit = limit + GameConfig.RUSHED_RATE;
        if (roll < limit) {
            rushedArrived = true;
            return new RushedCustomer(null);
        }
        limit = limit + GameConfig.PICKY_RATE;
        if (roll < limit) {
            pickyArrived = true;
            return new PickyCustomer(null);
        }
        limit = limit + GameConfig.GENEROUS_RATE;
        if (roll < limit) {
            return new GenerousCustomer(null);
        }

        if (!childrenAllowed) {
            return new NormalCustomer(null);
        }
        return new ChildCustomer(null);
    }
}

package com.happyhour.model.customer.state;

import com.happyhour.model.customer.Customer;

public class DrinkingState extends CustomerState {
    private double timeLeft;

    public DrinkingState(double duration) {
        timeLeft = duration;
    }

    @Override
    public void update(Customer customer, double deltaTime) {
        timeLeft = timeLeft - deltaTime;
        if (timeLeft <= 0) {
            customer.setState(new LeftState(false));
        }
    }
}

===== src/main/java/com/happyhour/model/drink/CocktailGrading.java =====
package com.happyhour.model.drink;

import java.util.ArrayList;

public class CocktailGrading implements GradingStrategy {

    @Override
    public int grade(Drink drink, Preparation preparation) {
        ArrayList<Ingredient> expected = drink.getIngredients();
        ArrayList<Ingredient> made = preparation.getIngredients();

        int missing = 0;
        for (Ingredient ingredient : expected) {
            if (!made.contains(ingredient)) {
                missing = missing + 1;
            }
        }
        int extra = made.size() - (expected.size() - missing);

        int score = 4 - missing - extra;
        if (score < 0) {
            score = 0;
        }

        if (isInOrder(expected, made)) {
            score = score + 2;
        }
        if (preparation.getGlass() == drink.getGlass()) {
            score = score + 2;
        }
        if (preparation.getTechnique() == drink.getTechnique()) {
            score = score + 2;
        }
        return score;
    }

    private boolean isInOrder(ArrayList<Ingredient> expected, ArrayList<Ingredient> made) {
        int lastIndex = -1;
        for (Ingredient ingredient : made) {
            int index = expected.indexOf(ingredient);
            if (index != -1) {
                if (index < lastIndex) {
                    return false;
                }
                lastIndex = index;
            }
        }
        return true;
    }
}

package com.happyhour.model.game;

import java.util.ArrayList;
import java.util.Random;

import com.happyhour.model.GameConfig;
import com.happyhour.model.customer.ChildCustomer;
import com.happyhour.model.customer.Customer;
import com.happyhour.model.customer.CustomerFactory;
import com.happyhour.model.customer.PickyCustomer;
import com.happyhour.model.drink.Drink;
import com.happyhour.model.drink.DrinkCategory;
import com.happyhour.model.drink.Glass;
import com.happyhour.model.drink.Ingredient;
import com.happyhour.model.drink.Preparation;
import com.happyhour.model.drink.RecipeBook;
import com.happyhour.model.drink.Technique;
import com.happyhour.model.score.CashRegister;

public class GameSession {
    private Customer[] seats = new Customer[GameConfig.SEATS];
    private Preparation preparation = new Preparation();
    private ArrayList<GameListener> listeners = new ArrayList<>();
    private CustomerFactory factory = new CustomerFactory();
    private OrderGenerator orders = new OrderGenerator();
    private RecipeBook recipeBook = new RecipeBook();
    private CashRegister register = new CashRegister();
    private int lostCustomers = 0;
    private double elapsed = 0;
    private Phase phase = Phase.OPENING;
    private double spawnTimer = 0;
    private double nextSpawnDelay;
    private boolean closed = false;
    private boolean paused = false;
    private boolean over = false;
    private boolean defeat = false;
    private Random rand = new Random();
    private String feedback = null;
    private int feedbackGrade = 0;
    private double feedbackTimer = 0;
    private boolean policeRaidInProgress = false;
    private double policeRaidTimer = 0;
    private boolean policeArrivalInProgress = false;
    private double policeArrivalTimer = 0;

    public GameSession() {
        nextSpawnDelay = randomBetween(GameConfig.FIRST_CUSTOMER_MIN, GameConfig.FIRST_CUSTOMER_MAX);
    }

    public void addListener(GameListener listener) {
        listeners.add(listener);
    }

    public void chooseGlass(Glass glass) {
        if (isPoliceRaidActive()) {
            return;
        }
        preparation.chooseGlass(glass);
    }

    public void addIngredient(Ingredient ingredient) {
        if (isPoliceRaidActive()) {
            return;
        }
        preparation.addIngredient(ingredient);
    }

    public void useTechnique(Technique technique) {
        if (isPoliceRaidActive()) {
            return;
        }
        preparation.useTechnique(technique);
    }

    public void placeCustomer(int seat, Customer customer) {
        seats[seat] = customer;
        for (GameListener listener : listeners) {
            listener.onCustomerArrived(seat, customer);
        }
    }

    private void removeAllCustomers() {
        for (int i = 0; i < GameConfig.SEATS; i++) {
            if (seats[i] != null) {
                freeSeat(i);
            }
        }
    }

    public boolean isPoliceRaidActive() {
        return policeRaidInProgress || policeArrivalInProgress;
    }

    public void serve(int seat) {
        Customer customer = seats[seat];
        if (over || closed || paused || isPoliceRaidActive()) {
            return;
        }
        if (customer == null || !customer.canBeServed() || !preparation.isReady()) {
            return;
        }

        if (customer.isChild()) {
            policeRaid();
            return;
        }

        Drink order = customer.getOrder();
        int grade = order.grade(preparation);
        Preparation served = preparation;
        preparation = new Preparation();
        showFeedback(grade);

        if (!customer.accepts(grade)) {
            loseCustomer(seat);
            return;
        }

        double speedBonus = 0;
        if (customer.isServedQuickly()) {
            speedBonus = GameConfig.SPEED_BONUS;
        }
        double tip = customer.computeTip(order.getBaseTip(grade), speedBonus);
        double price = order.getPrice() * grade / 10.0;
        register.recordSale(order, price, tip);
        for (GameListener listener : listeners) {
            listener.onCustomerServed(seat, price + tip);
        }
        customer.setServedPreparation(served);
        customer.serve();
        for (GameListener listener : listeners) {
            listener.onScoreChanged(register.getTotal());
        }
    }

    private void showFeedback(int grade) {
        if (grade == 10) {
            feedback = "PARFAIT ! 10/10";
        } else if (grade >= 5) {
            feedback = grade + "/10 : " + (grade * 10) + " % DU PRIX";
        } else {
            feedback = "NUL ! " + grade + "/10";
        }
        feedbackGrade = grade;
        feedbackTimer = GameConfig.FEEDBACK_TIME;
    }

    public void kick(int seat) {
        if (isPoliceRaidActive()) {
            return;
        }
        Customer customer = seats[seat];
        if (over || paused || customer == null) {
            return;
        }
        customer.kick();
        if (customer.isChild()) {
            freeSeat(seat);
        } else {
            loseCustomer(seat);
        }
    }

    public void togglePause() {
        if (!over && !isPoliceRaidActive()) {
            paused = !paused;
        }
    }

    public void spawnChild() {
        if (isPoliceRaidActive()) {
            return;
        }
        int seat = findFreeSeat();
        if (!over && !closed && seat != -1) {
            Customer child = new ChildCustomer(null);
            giveOrder(child);
            placeCustomer(seat, child);
        }
    }

    public void spawnCustomer() {
        if (isPoliceRaidActive()) {
            return;
        }
        int seat = findFreeSeat();
        if (!over && !closed && seat != -1) {
            Customer customer = factory.createCustomer(elapsed, phase != Phase.OPENING);
            giveOrder(customer);
            placeCustomer(seat, customer);
        }
    }

    public void skipPhase() {
        if (isPoliceRaidActive()) {
            return;
        }
        if (phase == Phase.OPENING) {
            elapsed = GameConfig.HAPPY_HOUR_START;
        } else if (phase == Phase.HAPPY_HOUR) {
            elapsed = GameConfig.RUSH_START;
        } else {
            elapsed = GameConfig.EVENING_END;
        }
    }

    public void countdownSkip() {
        if (isPoliceRaidActive()) {
            return;
        }
        elapsed = GameConfig.COUNTDOWN_SKIP_TIME;
    }

    public void update(double deltaTime) {
        if (over) {
            return;
        }

        if (policeRaidInProgress) {
            policeRaidTimer = policeRaidTimer - deltaTime;

            if (policeRaidTimer <= 0) {
                policeRaidInProgress = false;
                policeArrivalInProgress = true;

                for (GameListener listener : listeners) {
                    listener.onPoliceRaid();
                }

                policeArrivalTimer = GameConfig.POLICE_ARRIVAL_TIME;
            }
            return;
        }

        if (policeArrivalInProgress) {
            policeArrivalTimer = policeArrivalTimer - deltaTime;

            if (policeArrivalTimer <= 0) {
                policeArrivalInProgress = false;
                endEvening(true);
            }
            return;
        }

        if (paused) {
            return;
        }

        if (feedbackTimer > 0) {
            feedbackTimer = feedbackTimer - deltaTime;
            if (feedbackTimer <= 0) {
                feedback = null;
            }
        }

        elapsed = elapsed + deltaTime;
        Phase newPhase = getPhase();
        if (newPhase != phase) {
            phase = newPhase;
            for (GameListener listener : listeners) {
                listener.onPhaseChanged(phase);
            }
        }

        if (!closed && elapsed >= GameConfig.EVENING_END) {
            closeBar();
        }

        for (int i = 0; i < GameConfig.SEATS; i++) {
            Customer customer = seats[i];
            if (customer != null) {
                if (phase == Phase.RUSH) {
                    customer.setPatienceSpeed(GameConfig.RUSH_PATIENCE_SPEED);
                }
                customer.update(deltaTime);
                if (customer.hasLeft()) {
                    if (customer.isLost()) {
                        loseCustomer(i);
                    } else {
                        freeSeat(i);
                    }
                }
            }
        }

        if (closed) {
            if (!over && isBarEmpty()) {
                endEvening(false);
            }
            return;
        }

        spawnTimer = spawnTimer + deltaTime;
        if (spawnTimer >= nextSpawnDelay) {
            spawnTimer = 0;
            nextSpawnDelay = randomSpawnDelay();
            spawnCustomer();
        }
    }

    private void giveOrder(Customer customer) {
        boolean picky = customer instanceof PickyCustomer;
        DrinkCategory category = orders.pickCategory(phase, picky);
        customer.setOrder(recipeBook.randomDrink(category));
    }

    public Phase getPhase() {
        if (elapsed < GameConfig.HAPPY_HOUR_START) {
            return Phase.OPENING;
        } else if (elapsed < GameConfig.RUSH_START) {
            return Phase.HAPPY_HOUR;
        }
        return Phase.RUSH;
    }

    private double randomSpawnDelay() {
        if (phase == Phase.OPENING) {
            return randomBetween(GameConfig.OPENING_SPAWN_MIN, GameConfig.OPENING_SPAWN_MAX);
        } else if (phase == Phase.HAPPY_HOUR) {
            return randomBetween(GameConfig.HAPPY_HOUR_SPAWN_MIN, GameConfig.HAPPY_HOUR_SPAWN_MAX);
        }
        return randomBetween(GameConfig.RUSH_SPAWN_MIN, GameConfig.RUSH_SPAWN_MAX);
    }

    private double randomBetween(double min, double max) {
        return min + rand.nextDouble() * (max - min);
    }

    private void closeBar() {
        closed = true;
        for (int i = 0; i < GameConfig.SEATS; i++) {
            if (seats[i] != null && seats[i].canBeServed()) {
                freeSeat(i);
            }
        }
    }

    private boolean isBarEmpty() {
        for (int i = 0; i < GameConfig.SEATS; i++) {
            if (seats[i] != null) {
                return false;
            }
        }
        return true;
    }

    private void endEvening(boolean lost) {
        over = true;
        defeat = lost;
        for (GameListener listener : listeners) {
            listener.onEveningOver(lost);
        }
    }

    private void policeRaid() {
        if (isPoliceRaidActive()) {
            return;
        }

        policeRaidInProgress = true;
        policeRaidTimer = GameConfig.POLICE_DELAY;

        removeAllCustomers();
    }

    private void loseCustomer(int seat) {
        lostCustomers = lostCustomers + 1;
        freeSeat(seat);
        if (lostCustomers >= GameConfig.MAX_LOST) {
            endEvening(true);
        }
    }

    private void freeSeat(int seat) {
        if (seats[seat] != null) {
            Customer customer = seats[seat];
            seats[seat] = null;
            for (GameListener listener : listeners) {
                listener.onCustomerLeft(seat, customer);
            }
        }
    }

    private int findFreeSeat() {
        ArrayList<Integer> openSeats = new ArrayList<>();
        for (int i = 0; i < GameConfig.SEATS; i++) {
            if (seats[i] == null) {
                openSeats.add(i);
            }
        }
        if (openSeats.isEmpty()) {
            return -1;
        }
        return openSeats.get(rand.nextInt(openSeats.size()));
    }

    public Customer getCustomer(int seat) {
        return seats[seat];
    }

    public Drink getOrder(int seat) {
        Customer customer = seats[seat];
        if (customer == null || !customer.canBeServed()) {
            return null;
        }
        return customer.getOrder();
    }

    public ArrayList<Integer> getOccupiedSeats() {
        ArrayList<Integer> occupied = new ArrayList<>();
        for (int seat = 0; seat < GameConfig.SEATS; seat++) {
            if (seats[seat] != null) {
                occupied.add(seat);
            }
        }
        return occupied;
    }

    public String getFeedback() {
        return feedback;
    }

    public int getFeedbackGrade() {
        return feedbackGrade;
    }

    public Preparation getPreparation() {
        return preparation;
    }

    public RecipeBook getRecipeBook() {
        return recipeBook;
    }

    public CashRegister getRegister() {

                return categories[i];
            }
        }
        return DrinkCategory.BEER;
    }

package com.happyhour.model.score;

import java.io.File;
import java.io.FileNotFoundException;
import java.io.FileWriter;
import java.io.IOException;
import java.util.ArrayList;
import java.util.Scanner;

public class ScoreBoard {
    public static final int MAX_ENTRIES = 10;

    private String filePath;
    private ArrayList<ScoreEntry> entries = new ArrayList<>();

    public ScoreBoard(String filePath) {
        this.filePath = filePath;
        load();
    }

    public void addScore(String name, double score, boolean defeat) {
        String cleanName = name.replace(";", "");
        ScoreEntry entry = new ScoreEntry(cleanName, score, defeat);

        int index = 0;
        while (index < entries.size() && entries.get(index).getScore() >= score) {
            index = index + 1;
        }
        entries.add(index, entry);

        if (entries.size() > MAX_ENTRIES) {
            entries.remove(entries.size() - 1);
        }
        save();
    }

    public ArrayList<ScoreEntry> getEntries() {
        return entries;
    }

    private void save() {
        try {
            FileWriter writer = new FileWriter(filePath);
            for (ScoreEntry entry : entries) {
                writer.write(entry.getName() + ";" + entry.getScore() + ";" + entry.isDefeat() + "\n");
            }
            writer.close();
        } catch (IOException e) {
            System.out.println("Impossible de sauvegarder les scores.");
        }
    }

    private void load() {
        File file = new File(filePath);
        if (!file.exists()) {
            return;
        }
        try {
            Scanner reader = new Scanner(file);
            while (reader.hasNextLine()) {
                String[] parts = reader.nextLine().split(";");
                String name = parts[0];
                double score = Double.parseDouble(parts[1]);
                boolean defeat = Boolean.parseBoolean(parts[2]);
                entries.add(new ScoreEntry(name, score, defeat));
            }
            reader.close();
        } catch (FileNotFoundException e) {
            System.out.println("Impossible de lire les scores.");
        }
    }
}

package com.happyhour.view.game;

import java.util.ArrayList;

import javafx.scene.paint.Color;

// Dessine les verres (commandes, verre de Melvin, verre servi) avec la couleur de leur contenu
public class DrinkPainter {
    private GraphicsContext gc;
    private RecipeBook recipeBook;
    private HashMap<String, Image> tintCache = new HashMap<>();
    private Image beerFoam;
    private Image beerGlass;
    private Image beer;
    private Image highballGlass;
    private Image highball;
    private Image martiniGlass;
    private Image martini;
    private Image wineGlass;
    private Image wine;

    public DrinkPainter(GraphicsContext gc, RecipeBook recipeBook) {
        this.gc = gc;
        this.recipeBook = recipeBook;
        beerFoam = load("beerFoam.png", 100);
        beerGlass = load("beerGlass.png", 100);
        beer = load("beer.png", 100);
        highballGlass = load("highballGlass.png", 60);
        highball = load("highball.png", 60);
        martiniGlass = load("martiniGlass.png", 90);
        martini = load("martini.png", 90);
        wineGlass = load("wineGlass.png", 70);
        wine = load("wine.png", 70);
    }

    private Image load(String file, double width) {
        return new Image(getClass().getResourceAsStream("/images/" + file), width, 150, true, false);
    }

    public void drawDrink(Drink order, double centerX, double topY, double scale) {
        Color color;
        if (order.getGlass() == Glass.PINT) {
            color = getIngredientColor(Ingredient.BEER);
        } else if (order.getGlass() == Glass.WINE_GLASS) {
            color = getIngredientColor(Ingredient.WHITE_WINE);
        } else {
            color = getDrinkColor(order.getName());
        }
        double height = 150 * scale;
        drawGlass(order.getGlass(), color, order.getGlass() == Glass.PINT, centerX, topY + height / 2, scale);
    }

    public void drawPreparation(Preparation preparation, double centerX, double centerY, double scale) {
        Glass glass = preparation.getGlass();
        if (glass == null) {
            return;
        }
        ArrayList<Ingredient> liquids = getLiquids(preparation.getIngredients());
        Color color = null;
        if (!liquids.isEmpty()) {
            Drink match = findMatchingDrink(preparation);
            if (match != null && (glass == Glass.HIGHBALL || glass == Glass.MARTINI)) {
                color = getDrinkColor(match.getName());
            } else {
                color = mixColors(liquids);
            }
        }
        boolean foam = glass == Glass.PINT && preparation.getIngredients().contains(Ingredient.BEER);
        drawGlass(glass, color, foam, centerX, centerY, scale);
    }

    // Dessine un verre centré sur (centerX, centerY), avec son liquide si color n'est pas null
    private void drawGlass(Glass glass, Color color, boolean foam, double centerX, double centerY, double scale) {
        double width = 100 * scale;
        if (glass == Glass.WINE_GLASS) {
            width = 80 * scale;
        } else if (glass == Glass.HIGHBALL) {
            width = 60 * scale;
        }
        double height = 150 * scale;
        double x = centerX - width / 2;
        double y = centerY - height / 2;

        if (color != null) {
            gc.drawImage(getTintedLiquid(glass, color), x, y, width, height);
        }
        gc.drawImage(getGlassImage(glass), x, y, width, height);
        if (foam) {
            gc.drawImage(beerFoam, x, y, width, height);
        }
    }

    private Image getGlassImage(Glass glass) {
        if (glass == Glass.PINT) {
            return beerGlass;
        } else if (glass == Glass.WINE_GLASS) {
            return wineGlass;
        } else if (glass == Glass.HIGHBALL) {
            return highballGlass;
        }
        return martiniGlass;
    }

    private ArrayList<Ingredient> getLiquids(ArrayList<Ingredient> ingredients) {
        ArrayList<Ingredient> liquids = new ArrayList<>();
        for (Ingredient ingredient : ingredients) {
            if (ingredient.isLiquid()) {
                liquids.add(ingredient);
            }
        }
        return liquids;
    }

    private Drink findMatchingDrink(Preparation preparation) {
        ArrayList<Ingredient> made = getLiquids(preparation.getIngredients());
        for (Drink drink : recipeBook.getDrinks()) {
            if (drink.getGlass() == preparation.getGlass()
                    && getLiquids(drink.getIngredients()).equals(made)) {
                return drink;
            }
        }
        return null;
    }

    private Image getLiquidImage(Glass glass) {
        if (glass == Glass.PINT) {
            return beer;
        } else if (glass == Glass.WINE_GLASS) {
            return wine;
        } else if (glass == Glass.HIGHBALL) {
            return highball;
        }
        return martini;
    }

    private Color getDrinkColor(String name) {
        switch (name) {
            case "Gin Tonic":
                return Color.web("#00c7e6");
            case "Tequila Tonic":
                return Color.web("#ff9100");
            case "Pink Cape Cod":
                return Color.web("#FF7EB9");
            case "Margarita":
                return Color.web("#9BE15D");
            case "Clover Club":
                return Color.web("#cd07ff");
            case "Cosmopolitan":
                return Color.web("#D7263D");
        }
        return Color.WHITE;
    }

    private Color getIngredientColor(Ingredient ingredient) {
        switch (ingredient) {
            case VODKA:
                return Color.web("#e8f4ff");
            case GIN:
                return Color.web("#dff7f2");
            case TEQUILA:
                return Color.web("#f5d76e");
            case TRIPLE_SEC:
                return Color.web("#ffa733");
            case RASPBERRY_SYRUP:
                return Color.web("#e0115f");
            case TONIC:
                return Color.web("#c9f0ff");
            case LEMON_JUICE:
                return Color.web("#f7f06d");
            case CRANBERRY_JUICE:
                return Color.web("#b5122e");
            case BEER:
                return Color.web("#f2b134");
            case WHITE_WINE:
                return Color.web("#f3e9a6");
            default:
                return Color.WHITE;
        }
    }

    private Color mixColors(ArrayList<Ingredient> liquids) {
        double red = 0;
        double green = 0;
        double blue = 0;
        for (Ingredient ingredient : liquids) {
            Color color = getIngredientColor(ingredient);
            red = red + color.getRed();
            green = green + color.getGreen();
            blue = blue + color.getBlue();
        }
        int count = liquids.size();
        return Color.color(red / count, green / count, blue / count);
    }

    private Image getTintedLiquid(Glass glass, Color color) {
        String key = glass + color.toString();
        if (!tintCache.containsKey(key)) {
            tintCache.put(key, tintImage(getLiquidImage(glass), color));
        }
        return tintCache.get(key);
    }

    private WritableImage tintImage(Image source, Color tint) {
        int width = (int) source.getWidth();
        int height = (int) source.getHeight();
        PixelReader reader = source.getPixelReader();
        WritableImage result = new WritableImage(width, height);
        PixelWriter writer = result.getPixelWriter();
        for (int y = 0; y < height; y++) {
            for (int x = 0; x < width; x++) {
                Color color = reader.getColor(x, y);
                double brightness = color.getBrightness();
                Color tintedColor = Color.color(tint.getRed() * brightness, tint.getGreen() * brightness, tint.getBlue() * brightness, color.getOpacity());
                writer.setColor(x, y, tintedColor);
            }
        }
        return result;
    }
}

    private Image getSprite(Customer customer) {
        if (customer instanceof ChildCustomer) {
            return childSprite;
        } else if (customer instanceof RushedCustomer) {
            return rushedSprite;
        } else if (customer instanceof PickyCustomer) {
            return pickySprite;
        } else if (customer instanceof GenerousCustomer) {
            return generousSprite;
        }
        return normalSprite;
    }

    private ImageView createArmView(Translate pos, Rotate rot, boolean flipped) {
        ImageView view = new ImageView(arm);
        view.setX(-1410);
        view.setY(-170);
        view.setFitWidth(1580);
        view.setFitHeight(320);
        view.setSmooth(false);
        view.setMouseTransparent(true);
        if (flipped) {
            view.getTransforms().setAll(pos, rot, new Scale(1, -1));
        } else {
            view.getTransforms().setAll(pos, rot);
        }
        return view;
    }

    @Override
    public void onPoliceRaid() {
        for (int i = 0; i < 6; i++) {
            double x = seatX(i);
            double y = 390;
            CustomerAnimation animation = new CustomerAnimation(i, x, y, policeImage, i * 83L);
            policeAnimations.add(animation);
        }
    }

    @Override
    public void onCustomerArrived(int seat, Customer customer) {
        double x = seatX(seat);
        double y = 390;

        customerAnimations.add(CustomerAnimation.arrival(seat, getSprite(customer), x, y));
    }

    @Override
    public void onCustomerLeft(int seat, Customer customer) {
        double x = seatX(seat);
        double y = 390;

        customerAnimations.add(CustomerAnimation.departure(seat, getSprite(customer), x, y));
    }

    @Override
    public void onCustomerServed(int seat, double amount) {
        popupText[seat] = "+" + String.format("%.0f", amount) + " €";
        popupStart[seat] = System.currentTimeMillis();
        SoundManager.play("cash");
    }

    private void renderCustomers() {
        customerAnimations.removeIf(animation -> animation.getAge() >= 500);

        for (int i = 0; i < 6; i++) {
            Customer customer = session.getCustomer(i);

            if (customer != null && !isAnimated(i)) {
                double x = seatX(i);
                double y = 390;
                batch.drawImage(getSprite(customer), x, y, 160, 350);
            }
        }

        for (CustomerAnimation animation : policeAnimations) {
            animation.update();

            batch.save();
            batch.setGlobalAlpha(animation.getOpacity());
            batch.drawImage(
                animation.getCustomerImage(),
                animation.getX(),
                animation.getY(),
                160,
                350
            );
            batch.restore();
        }

        for (CustomerAnimation animation : customerAnimations) {
            animation.update();

            batch.save();
            batch.setGlobalAlpha(animation.getOpacity());
            batch.drawImage(
                animation.getCustomerImage(),
                animation.getX(),
                animation.getY(),
                160,
                350
            );
            batch.restore();
        }
    }

    public static double seatX(int seat) {
        return 310 + seat * 240;
    }

    private boolean isAnimated(int seat) {
        for (CustomerAnimation animation : customerAnimations) {
            if (animation.getSeat() == seat) {
                return true;
            }
        }
        return false;
    }

    private void resizeGame(Pane gamePane, Pane root) {
        double scaleX = root.getWidth() / WIDTH;
        double scaleY = root.getHeight() / HEIGHT;

        double scale = Math.min(scaleX, scaleY);

        double x = (root.getWidth() - WIDTH * scale) / 2;
        double y = (root.getHeight() - HEIGHT * scale) / 2;

        Scale transformation = new Scale(scale, scale, 0, 0);

        gamePane.getTransforms().setAll(transformation);

        gamePane.setLayoutX(x);
        gamePane.setLayoutY(y);
    }

    private Label createHudLabel(Pos position) {
        Label label = new Label();
        label.setStyle("-fx-text-fill: #ffd9ec;"
                + "-fx-font-family: 'Monospaced';"
                + "-fx-font-weight: bold;"
                + "-fx-font-size: 26px;");
        label.setPrefWidth((WIDTH - 60 - 80 - 40) / 3.0);
        label.setAlignment(position);
        return label;
    }

    private void updateHud() {
        double remaining = GameConfig.EVENING_END - session.getElapsed();
        if (remaining < 0) {
            remaining = 0;
        }
        int minutes = (int) remaining / 60;
        int seconds = (int) remaining % 60;
        int secondsLeft = (int) Math.ceil(remaining);

        if (secondsLeft <= 10 && secondsLeft > 0
                && secondsLeft != lastCountdownSecond) {

            SoundManager.play("countdown");
            lastCountdownSecond = secondsLeft;
        }
        timeLabel.setText("TEMPS " + String.format("%d:%02d", minutes, seconds));
        moneyLabel.setText("CAISSE " + String.format("%.0f €", session.getRegister().getTotal()));
        lostLabel.setText("CLIENTS PERDUS " + session.getLostCustomers() + "/" + GameConfig.MAX_LOST);

        Preparation preparation = session.getPreparation();
        boolean currentHasGlass = preparation.getGlass() != null;

        if (currentHasGlass != hasGlass) {
            hasGlass = currentHasGlass;
            arm = melvin.getArm(hasGlass);
            arm1View.setImage(arm);
            arm2View.setImage(arm);
        }
        ArrayList<String> parts = new ArrayList<>();
        if (preparation.getGlass() != null) {
            parts.add(DrinkNames.glassName(preparation.getGlass()));
        }
        for (Ingredient ingredient : preparation.getIngredients()) {
            parts.add(DrinkNames.ingredientName(ingredient));
        }
        if (preparation.getTechnique() != Technique.NONE) {
            parts.add(DrinkNames.techniqueName(preparation.getTechnique()));
        }

        if (parts.isEmpty()) {
            prepLabel.setText("EN MAIN : RIEN");
        } else {
            prepLabel.setText("EN MAIN : " + String.join(" + ", parts).toUpperCase());
        }

        String feedback = session.getFeedback();
        if (feedback == null) {
            feedbackLabel.setText("");
        } else {
            String color = "#ffffff";
            if (session.getFeedbackGrade() == 10) {
                color = "#ff9900";
            } else if (session.getFeedbackGrade() < 5) {
                color = "#ff4d4d";
            }
            feedbackLabel.setStyle("-fx-text-fill: " + color + ";"
                    + "-fx-font-family: 'Monospaced';"
                    + "-fx-font-weight: bold;"
                    + "-fx-font-size: 48px;");
            feedbackLabel.setText(feedback);
        }
    }

    private void renderDisco() {
        if (!GameSettings.discoMode) {
            setEffect(null);
            lightsView.setEffect(null);
            return;
        }
        double time = System.currentTimeMillis() / 3000.0;
        discoHue.setHue(0.16 + Math.sin(time) * 0.5);
        setEffect(discoHue);
        lightsView.setEffect(discoHue);
    }

    public void moveHover(String text, MouseEvent event) {
        Point2D point = gamePane.sceneToLocal(event.getSceneX(), event.getSceneY());
        double dx = point.getX() - hoverX;
        double dy = point.getY() - hoverY;
        if (dx * dx + dy * dy < 16) {
            return;
        }
        hoverLabel.setVisible(false);
        hoverLabel.setText(text);
        hoverX = point.getX();
        hoverY = point.getY();
        hoverTimer.playFromStart();
    }

    public void stopHover() {
        hoverTimer.stop();
        hoverLabel.setVisible(false);
    }

    public void installHover(Node node, String text) {
        node.addEventHandler(MouseEvent.MOUSE_MOVED, event -> moveHover(text, event));
        node.addEventHandler(MouseEvent.MOUSE_EXITED, event -> stopHover());
    }

    public void renderOrders() {
        drinkBatch.clearRect(0, 0, WIDTH, HEIGHT);
        for (int seat = 0; seat < GameConfig.SEATS; seat++) {
            Customer customer = session.getCustomer(seat);
            if (customer != null && customer.getOrder() != null) {
                double centerX = seatX(seat) + 80;
                if (customer.canBeServed()) {
                    drinkBatch.drawImage(orderBubble, seatX(seat) - 10, 170, 180, 260);
                    painter.drawDrink(customer.getOrder(), centerX, 190, 1);
                    drawPatienceBar(customer, seat);
                } else if (customer.getServedPreparation() != null) {
                    painter.drawPreparation(customer.getServedPreparation(), centerX, 675, 0.6);
                }
            }
        }
        drawMoneyPopups();
    }

    private void drawPatienceBar(Customer customer, int seat) {
        double ratio = customer.getPatience() / customer.getMaxPatience();
        if (ratio < 0) {
            ratio = 0;
        }
        double x = seatX(seat) + 10;
        double y = 150;
        double width = 140;
        double height = 14;

        Color color = Color.web("#9BE15D");
        if (ratio < 0.25) {
            color = Color.web("#ff4d4d");
        } else if (ratio < 0.5) {
            color = Color.web("#ff9900");
        }

        drinkBatch.setFill(Color.BLACK);
        drinkBatch.fillRect(x - 3, y - 3, width + 6, height + 6);
        drinkBatch.setFill(Color.web("#531432"));
        drinkBatch.fillRect(x, y, width, height);
        drinkBatch.setFill(color);
        drinkBatch.fillRect(x, y, width * ratio, height);
    }

    private void drawMoneyPopups() {
        long now = System.currentTimeMillis();
        drinkBatch.setFont(Font.font("Monospaced", FontWeight.BOLD, 44));
        drinkBatch.setTextAlign(TextAlignment.CENTER);
        for (int seat = 0; seat < GameConfig.SEATS; seat++) {
            if (popupText[seat] != null) {
                double age = now - popupStart[seat];
                if (age > POPUP_TIME) {
                    popupText[seat] = null;
                } else {
                    double progress = age / POPUP_TIME;
                    double x = seatX(seat) + 80;
                    double y = 330 - progress * 100;
                    drinkBatch.setGlobalAlpha(1 - progress);
                    drinkBatch.setFill(Color.BLACK);
                    drinkBatch.fillText(popupText[seat], x + 3, y + 3);
                    drinkBatch.setFill(Color.web("#ff9900"));
                    drinkBatch.fillText(popupText[seat], x, y);
                    drinkBatch.setGlobalAlpha(1);
                }
            }
        }
    }

    public void showKick(int seat) {
        explosions.add(new Explosion(seatX(seat) - 120, 340));
    }

    public void pause() {
        if (session.isOver()) {
            return;
        }
        if (!session.isPaused()) {
            session.togglePause();
        }
        stopHover();
        new PauseScreen(app, this).start(stage);
    }

    public void openSettings() {
        if (session.isOver() || session.isPoliceRaidActive()) {
            return;
        }
        if (!session.isPaused()) {
            session.togglePause();
        }
        stopHover();
        new SettingsScreen(app, () -> resume()).start(stage);
    }

    public void resume() {
        stage.getScene().setRoot(root);
        controller.installKeys(stage.getScene());
        if (session.isPaused()) {
            session.togglePause();
        }
    }

    public void restart() {
        gameLoop.stop();
        app.showGame();
    }

    public void backToMenu() {
        gameLoop.stop();
        app.showMenu();
    }

    public void quit() {
        gameLoop.stop();
        Platform.exit();
    }

    public void endGame(GameSession session) {
        app.showSummary(session);
    }

    private void renderPoliceLights() {
        boolean raid = session.isPoliceRaidActive();
        policeLightLeft.setVisible(raid);
        policeLightRight.setVisible(raid);
        if (!raid) {
            return;
        }
        boolean redFlash = System.currentTimeMillis() % 1500 < 750;
        if (redFlash) {
            policeLightLeft.setOpacity(0);
            policeLightRight.setOpacity(0.7);
        } else {
            policeLightLeft.setOpacity(0.7);
            policeLightRight.setOpacity(0);
        }
    }

    public void render() {
        updateHud();
        renderDisco();
        renderPoliceLights();
        batch.clearRect(0, 0, WIDTH, HEIGHT);

        batch.drawImage(seats, 310, 590, 1360, 120);
        renderCustomers();
        batch.drawImage(table, 0, 540, WIDTH, 540);
        renderOrders();
        explosions.removeIf(explosion -> explosion.getAge() > 500);
        kickBatch.clearRect(0, 0, WIDTH, HEIGHT);
        for (Explosion explosion : explosions) {
            kickBatch.save();
            explosion.update();
            kickBatch.setGlobalAlpha(explosion.getOpacity());
            kickBatch.translate(explosion.getX() + 200, explosion.getY() + 200);
            kickBatch.scale(explosion.getScale(), explosion.getScale());
            kickBatch.rotate(explosion.getRotation());
            if (explosion.getFlipStatus() == 1) {
                kickBatch.scale(-1,1);
            }
            kickBatch.drawImage(kick, -200, -200, 400, 400);
            kickBatch.restore();
        }

        double angle = Math.atan2(melvin.getY() - 1280, melvin.getX() - 990) * (180 / Math.PI);

        arm1Pos.setX(melvin.getX() - 192);
        arm1Pos.setY(melvin.getY());
        arm1Rot.setAngle(angle);

        arm2Pos.setX(melvin.getX() + 192);
        arm2Pos.setY(melvin.getY());
        arm2Rot.setAngle(angle);

        painter.drawPreparation(session.getPreparation(), melvin.getX(), melvin.getY(), 1);
    }
}

package com.happyhour.view.game;
import javafx.scene.image.Image;

public class Player {
    private Image arm;
    private Image armClench;
    private double mouseX;
    private double mouseY;
    public Player() {
        arm = new Image(getClass().getResourceAsStream("/images/arm.png"),1580, 320, true, false);
        armClench = new Image(getClass().getResourceAsStream("/images/armClench.png"),1580, 320, true, false);
        mouseX = 0;
        mouseY = 0;
    }
    public double getX() {
        return mouseX;
    }
    public double getY() {
        return mouseY;
    }
    public Image getArm(boolean status) {
        if (status == false) {
            return arm;
        }
        else {
            return armClench;
        }
    }
    public void updateMouse(double x, double y) {
        mouseX = x;
        mouseY = y;
    }
}

package com.happyhour.view.screen;

import com.happyhour.HappyHourApp;

import javafx.geometry.Pos;
import javafx.scene.Scene;
import javafx.scene.control.Button;
import javafx.scene.control.Label;
import javafx.scene.control.TextField;
import javafx.scene.effect.DropShadow;
import javafx.scene.layout.VBox;
import javafx.scene.paint.Color;
import javafx.stage.Stage;

public abstract class Screen {
    private static final String FONT = "-fx-font-family: 'Monospaced'; -fx-font-weight: bold; ";
    private static final String BACKGROUND = "-fx-background-color: linear-gradient(to bottom, "
            + "#993366 0%, #993366 86%, black 86%, black 87.5%, #a52763 87.5%, #a52763 100%); "
            + "-fx-padding: 40 40 110 40;";
    private static final String BUTTON = FONT + "-fx-font-size: 22px; -fx-text-fill: black; "
            + "-fx-background-color: #ff9900; -fx-background-radius: 0; "
            + "-fx-border-color: black; -fx-border-width: 4; -fx-cursor: hand;";
    private static final String BUTTON_HOVER = FONT + "-fx-font-size: 22px; -fx-text-fill: black; "
            + "-fx-background-color: #ffb84d; -fx-background-radius: 0; "
            + "-fx-border-color: black; -fx-border-width: 4; -fx-cursor: hand;";
    private static final String SECONDARY = FONT + "-fx-font-size: 22px; -fx-text-fill: #bf7fbf; "
            + "-fx-background-color: #531432; -fx-background-radius: 0; "
            + "-fx-border-color: black; -fx-border-width: 4; -fx-cursor: hand;";
    private static final String SECONDARY_HOVER = FONT + "-fx-font-size: 22px; -fx-text-fill: #bf7fbf; "
            + "-fx-background-color: #6e1c44; -fx-background-radius: 0; "
            + "-fx-border-color: black; -fx-border-width: 4; -fx-cursor: hand;";

    protected HappyHourApp app;

    public Screen(HappyHourApp app) {
        this.app = app;
    }

    protected VBox createLayout(String title) {
        VBox root = new VBox(20);
        root.setAlignment(Pos.CENTER);
        root.setStyle(BACKGROUND);

        Label titleLabel = new Label(title.toUpperCase());
        titleLabel.setStyle(FONT + "-fx-font-size: 64px; -fx-text-fill: #ff9900;");
        titleLabel.setEffect(new DropShadow(0, 5, 5, Color.BLACK));
        root.getChildren().add(titleLabel);
        return root;
    }

    protected Label createSubtitle(String text) {
        Label label = new Label(text);
        label.setStyle(FONT + "-fx-font-size: 22px; -fx-text-fill: white;");
        return label;
    }

    protected VBox createCard() {
        VBox card = new VBox(10);
        card.setAlignment(Pos.CENTER_LEFT);
        card.setMaxWidth(900);
        card.setStyle("-fx-background-color: #531432; -fx-border-color: black; "
                + "-fx-border-width: 4; -fx-padding: 20 35 20 35;");
        card.setEffect(new DropShadow(0, 6, 6, Color.BLACK));
        return card;
    }

    protected Label createText(String text) {
        Label label = new Label(text);
        label.setStyle(FONT + "-fx-font-size: 20px; -fx-text-fill: white;");
        label.setWrapText(true);
        return label;
    }

    protected Label createHighlight(String text) {
        Label label = createText(text);
        label.setStyle(FONT + "-fx-font-size: 20px; -fx-text-fill: #ff9900;");
        return label;
    }

    protected Button createButton(String text) {
        return makeButton(text, BUTTON, BUTTON_HOVER);
    }

    protected Button createSecondaryButton(String text) {
        return makeButton(text, SECONDARY, SECONDARY_HOVER);
    }

    private Button makeButton(String text, String style, String hoverStyle) {
        Button button = new Button(text);
        button.setPrefWidth(280);
        button.setStyle(style);
        button.setEffect(new DropShadow(0, 5, 5, Color.BLACK));
        button.setOnMouseEntered(event -> {
            button.setStyle(hoverStyle);
            button.setTranslateX(-2);
            button.setTranslateY(-2);
        });
        button.setOnMouseExited(event -> {
            button.setStyle(style);
            button.setTranslateX(0);
            button.setTranslateY(0);
        });
        return button;
    }

    protected TextField createTextField(String placeholder) {
        TextField field = new TextField();
        field.setPromptText(placeholder);
        field.setMaxWidth(280);
        field.setStyle(FONT + "-fx-font-size: 20px; -fx-background-radius: 0; "
                + "-fx-border-color: black; -fx-border-width: 4;");
        return field;
    }

    protected void display(Stage stage, VBox root) {
        if (stage.getScene() == null) {
            stage.setScene(new Scene(root, 1280, 720));
        } else {
            stage.getScene().setRoot(root);
            stage.getScene().setOnKeyPressed(null);
        }
    }

    public abstract void start(Stage stage);
}

            + "-fx-font-size: 26px;";

        soundBox.setOnAction(event -> SoundManager.setMuted(!soundBox.isSelected()));

        Label sizeLabel = new Label("TAILLE DE L'ÉCRAN");
        sizeLabel.setStyle(TEXT_STYLE);
        ComboBox<String> sizeBox = new ComboBox<>();
        sizeBox.getItems().addAll("Plein écran", "Fenêtre agrandie", "1920 x 1080", "1600 x 900", "1280 x 720");
        sizeBox.setValue(GameSettings.screenSize);
        sizeBox.setStyle("-fx-font-family: 'Monospaced'; -fx-font-weight: bold; -fx-font-size: 20px;");
        sizeBox.setOnAction(event -> {
            GameSettings.screenSize = sizeBox.getValue();
            applyScreenSize(stage, sizeBox.getValue());
        });

        CheckBox discoBox = new CheckBox("MODE DISCO");
        discoBox.setStyle(TEXT_STYLE);
        discoBox.setSelected(GameSettings.discoMode);
        discoBox.setOnAction(event -> {
            GameSettings.discoMode = discoBox.isSelected();
            app.switchMusic(GameSettings.discoMode);
        });

        Button backButton = createButton("Retour");
        backButton.setOnAction(event -> onBack.run());

        root.getChildren().addAll(musicLabel, musicSlider, soundBox, sizeLabel, sizeBox, discoBox, backButton);
        display(stage, root);

    private void applyScreenSize(Stage stage, String choice) {
        stage.setFullScreen(false);
        stage.setMaximized(false);
        if (choice.equals("Plein écran")) {
            stage.setFullScreen(true);
        } else if (choice.equals("Fenêtre agrandie")) {
            stage.setMaximized(true);
        } else {
            String[] parts = choice.split(" x ");
            stage.setWidth(Integer.parseInt(parts[0]));
            stage.setHeight(Integer.parseInt(parts[1]));
            stage.centerOnScreen();
        }
    }

package com.happyhour.view.screen;

import com.happyhour.HappyHourApp;

import javafx.event.EventHandler;
import javafx.geometry.Pos;
import javafx.scene.control.Button;
import javafx.scene.image.Image;
import javafx.scene.image.ImageView;
import javafx.scene.input.KeyCode;
import javafx.scene.input.KeyEvent;
import javafx.scene.layout.HBox;
import javafx.scene.layout.VBox;
import javafx.stage.Stage;

public class TutorialScreen extends Screen {
    private static final int PAGES = 7;
    private int page = 1;
    private Stage stage;
    private ImageView pageView;
    private Button previousButton;
    private Button nextButton;
    private EventHandler<KeyEvent> arrowKeys;

    public TutorialScreen(HappyHourApp app) {
        super(app);
    }

    @Override
    public void start(Stage stage) {
        this.stage = stage;
        VBox root = new VBox(15);
        root.setAlignment(Pos.CENTER);
        root.setStyle("-fx-background-color: #993366; -fx-padding: 20;");

        pageView = new ImageView();
        pageView.setPreserveRatio(true);
        pageView.fitWidthProperty().bind(stage.widthProperty().multiply(0.95));
        pageView.fitHeightProperty().bind(stage.heightProperty().multiply(0.82));

        previousButton = createSecondaryButton("Précédent");
        previousButton.setOnAction(event -> previousPage());

        Button menuButton = createSecondaryButton("Menu");
        menuButton.setOnAction(event -> leave());

        nextButton = createButton("Suivant");
        nextButton.setOnAction(event -> nextPage());

        HBox buttons = new HBox(20, previousButton, menuButton, nextButton);
        buttons.setAlignment(Pos.CENTER);

        root.getChildren().addAll(pageView, buttons);
        display(stage, root);

        arrowKeys = event -> {
            if (event.getCode() == KeyCode.RIGHT) {
                nextPage();
                event.consume();
            } else if (event.getCode() == KeyCode.LEFT) {
                previousPage();
                event.consume();
            }
        };
        stage.getScene().addEventFilter(KeyEvent.KEY_PRESSED, arrowKeys);

        showPage();
    }

    private void showPage() {
        pageView.setImage(new Image(getClass().getResourceAsStream("/images/tutorialscreen/tuto" + page + ".png")));
        previousButton.setDisable(page == 1);
        if (page == PAGES) {
            nextButton.setText("Terminer");
        } else {
            nextButton.setText("Suivant");
        }
    }

    private void nextPage() {
        if (page < PAGES) {
            page = page + 1;
            showPage();
        } else {
            leave();
        }
    }

    private void previousPage() {
        if (page > 1) {
            page = page - 1;
            showPage();
        }
    }

    private void leave() {
        stage.getScene().removeEventFilter(KeyEvent.KEY_PRESSED, arrowKeys);
        app.showMenu();
    }
}
